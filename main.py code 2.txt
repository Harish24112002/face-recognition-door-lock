# Brightness adjustment function
def adjust_brightness_contrast(image, brightness=0, contrast=0):
    """Adjusts brightness and contrast of an image.

    Args:
        image: Input image (NumPy array).
        brightness: Brightness adjustment (integer, -255 to 255).
        contrast: Contrast adjustment (float, typically 0.0 to 3.0).

    Returns:
        Adjusted image (NumPy array).
    """
    adjusted = cv.convertScaleAbs(image, alpha=contrast, beta=brightness)
    return adjusted
