# Text-to-Speech

# Image Preprocessing

# Load Face Classifier

# Load Face Encodings

# Start Video Stream

fps = FPS().start()

# Face Recognition Threshold
FACE_RECOGNITION_THRESHOLD = 0.5  

def speak(text):

    engine.runAndWait()

while True:
    frame = vs.read()
    frame = imutils.resize(frame, width=500)
    gray = cv.cvtColor(frame, cv.COLOR_BGR2GRAY)

    # Face Detection
    faces = faceClassifier.detectMultiScale(gray, scaleFactor=1.2, minNeighbors=6)

    # Face Recognition
    boxes = face_recognition.face_locations(frame)
    encodings = face_recognition.face_encodings(frame, boxes)
    names = []

    unknown_detected = False  # Flag to track unknown persons

    for encoding in encodings:
        matches = face_recognition.compare_faces(data["encodings"], encoding)
        face_distances = face_recognition.face_distance(data["encodings"], encoding)
        
        name = "Unknown"
        if True in matches:
            best_match_index = np.argmin(face_distances)
            if face_distances[best_match_index] < FACE_RECOGNITION_THRESHOLD:
                name = data["name"][best_match_index]  # Corrected key 'name' here

        names.append(name)
        time.sleep(12)
        if name == "Unknown":
            unknown_detected = True

    # Draw Bounding Box
    for ((top, right, bottom, left), name) in zip(boxes, names):
        color = (0, 255, 0) if name != "Unknown" else (0, 0, 255)
        cv.rectangle(frame, (left, top), (right, bottom), color, 2)
        cv.putText(frame, name, (left, top - 10), cv.FONT_HERSHEY_SIMPLEX, 0.8, color, 2)

    for name in names:
        if name == "Unknown":
            print("UNKNOWN PERSON here")
            # time.sleep(30)
            files = "output/unknown.jpg"

            speak("Unknown person detected.")

            #arduino.write(b"A")
            #arduino.write(b"B")
        else:
